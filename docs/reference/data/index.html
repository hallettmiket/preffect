    Reading and preparing data¶

    Loading counts, checking they can be used, choosing genes and samples, hiding entries for an imputation experiment, and reading the sample-sample network.

    The checks are worth running before anything else. preffect check is built from them, and on real data they catch things worth knowing: repeated gene names, a validation set that is a copy of the training set, and a network with no edges in it.

    Loading¶

    preffect.data.loading ¶

    Finding and reading the count files for a run.

    One folder holds one split. Inside it, one file per tissue, named so that it ends in tau_<n>.h5ad where <n> is the tissue number counting from 1. A single-tissue run has one file ending tau_1.h5ad.

    Two deliberate choices, both from defects in version 0.1:

    The search does not descend into subfolders. Version 0.1 used a non-recursive listing too, which was correct, but by accident rather than by decision: several real dataset folders contain an old/ subfolder holding previous versions of the same files, and a recursive search would silently pick them up. The behaviour is kept and now it is written down.

    The number of tissues found must match the number asked for. Version 0.1 quietly changed the requested number to whatever it found, and the message it logged about doing so printed the already-corrected value, so it was wrong in both directions at once.

    def find_tissue_files(folder: Path, n_tissues: int) -> list[Path]:
    """Return one file per tissue, ordered by tissue number.

    :param folder: the folder holding one split, for example the training set.
    :param n_tissues: how many tissues to expect.
    :raises DataError: if the folder is missing, or the files found do not
        number exactly ``n_tissues`` with tissue numbers 1 to ``n_tissues``.
    """
    folder = Path(folder)
    if not folder.is_dir():
        raise DataError(
            f"no such folder: {folder}. It should hold one '*tau_<n>.h5ad' file "
            f"per tissue."
        )

    found: dict[int, Path] = {}
    duplicates: dict[int, list[Path]] = {}
    for entry in sorted(folder.iterdir()):
        if not entry.is_file():
            continue
        match = TISSUE_FILE_PATTERN.search(entry.name)
        if match is None:
            continue
        index = int(match.group(1))
        if index in found:
            duplicates.setdefault(index, [found[index]]).append(entry)
        else:
            found[index] = entry

    if duplicates:
        detail = "; ".join(
            f"tissue {i}: {', '.join(p.name for p in paths)}"
            for i, paths in sorted(duplicates.items())
        )
        raise DataError(
            f"more than one file for the same tissue in {folder}. {detail}. "
            f"Each tissue number may appear once."
        )

    if not found:
        others = [p.name for p in sorted(folder.iterdir()) if p.is_file()][:5]
        raise DataError(
            f"no files matching '*tau_<n>.h5ad' in {folder}. "
            + (f"The folder does contain: {', '.join(others)}." if others
               else "The folder is empty.")
            + " Note that subfolders are not searched."
        )

    expected = set(range(1, n_tissues + 1))
    actual = set(found)
    if actual != expected:
        missing = sorted(expected - actual)
        extra = sorted(actual - expected)
        parts = [f"{folder} was asked for {n_tissues} tissue(s)"]
        if missing:
            parts.append(f"no file for tissue number {missing}")
        if extra:
            parts.append(
                f"found tissue number {extra}, which is outside 1 to {n_tissues}"
            )
        parts.append(f"files present: {sorted(p.name for p in found.values())}")
        raise DataError(". ".join(parts) + ".")

    return [found[i] for i in range(1, n_tissues + 1)]

    def load_tissues(folder: Path, n_tissues: int) -> list[ad.AnnData]:
    """Read one split from disk, ordered by tissue number.

    :raises DataError: if a file cannot be read, naming which one.
    """
    tissues = []
    for path in find_tissue_files(folder, n_tissues):
        try:

        except Exception as exc:  # noqa: BLE001 - the cause is reported verbatim
            raise DataError(f"could not read {path}: {exc}") from exc
    return tissues

    Checks¶

    preffect.data.checks ¶

    Checks on count data, which stop a run rather than noting a problem and continuing.

    Version 0.1 had a function that looked like it validated its input. Every check inside it was written as "raise an error, catch that same error, write it to a log, carry on", and the function ended with an unconditional return True. The caller read::

    if not self.check_fidelity_of_anndata():
    raise ValueError('Something wrong with input data. Check log.')

    which could never fire. Malformed input was reported to a log file nobody opens and then used anyway. That is defect 16 in how_this_project_breaks.md.

    Here the checks collect every problem they find and raise once, so a researcher with a malformed file is told everything that is wrong with it in one go rather than discovering the faults one run at a time.

    @dataclass(frozen=True, slots=True)
class SplitOverlap:
    """How much two splits have in common."""

    n_train: int
    n_validation: int
    n_shared_names: int
    #: Of the samples sharing a name, how many also hold identical counts. This
    #: is the number that means anything. A shared name on its own does not:
    #: simulated data often numbers its samples '0', '1', '2', so two genuinely
    #: independent draws share every name while holding different counts. An
    #: earlier version of this check used names alone and reported the simple
    #: vignette, which is two independent draws, as overlapping.
    n_identical_rows: int

    @property
    def fraction_repeated(self) -> float:
        """How much of the training set reappears in validation, by value."""
        return self.n_identical_rows / self.n_train if self.n_train else 0.0

    @property
    def values_identical_on_shared(self) -> bool:
        """Every sample sharing a name also holds identical counts."""
        return self.n_shared_names > 0 and self.n_identical_rows == self.n_shared_names

    @property
    def is_effectively_identical(self) -> bool:
        """The two splits are the same data: same samples, same counts."""
        return (
            self.n_train > 0
            and self.n_identical_rows == self.n_train == self.n_validation
        )

    def problems_with_counts(adata: ad.AnnData, label: str) -> list[str]:
    """Return every problem with one tissue's counts, as readable sentences.

    An empty list means the data is usable.
    """
    problems: list[str] = []

    if adata.X is None:
        return [f"{label}: has no expression matrix"]

    matrix = adata.X
    dense = np.asarray(matrix.todense()) if hasattr(matrix, "todense") else np.asarray(matrix)

    if dense.ndim != 2:
        return [f"{label}: expression matrix has {dense.ndim} dimensions, expected 2"]

    n_samples, n_genes = dense.shape
    if n_samples == 0:
        problems.append(f"{label}: no samples")
    if n_genes == 0:
        problems.append(f"{label}: no genes")

    if dense.size:
        if np.isnan(dense).any():
            count = int(np.isnan(dense).sum())
            problems.append(
                f"{label}: {count} missing value(s) in the counts. A count model "
                f"cannot use them; set them to 0 or drop the gene or sample."
            )
        finite = dense[np.isfinite(dense)]
        if finite.size and finite.min() < 0:
            problems.append(
                f"{label}: smallest count is {finite.min()}, and counts cannot be "
                f"negative. This usually means the matrix holds transformed values "
                f"rather than raw counts."
            )
        if finite.size and not np.all(np.equal(np.mod(finite, 1), 0)):
            problems.append(
                f"{label}: counts are not whole numbers. PREFFECT models counts, so "
                f"normalised or log-transformed values will not fit."
            )

    problems.extend(_repeated_names(adata.obs_names, "sample", label))
    problems.extend(_repeated_names(adata.var_names, "gene", label))

    return problems

    def check_tissues(tissues: list[ad.AnnData], split: str = "data") -> None:
    """Check every tissue, and report all problems at once.

    :param split: what to call this set in the message, for example "training".
    :raises DataError: if any tissue has a problem.
    """
    problems: list[str] = []
    for index, adata in enumerate(tissues, start=1):
        problems.extend(problems_with_counts(adata, f"{split} tissue {index}"))

    gene_sets = [tuple(t.var_names) for t in tissues]
    if len(tissues) > 1 and len(set(gene_sets)) > 1:
        sizes = [len(g) for g in gene_sets]
        problems.append(
            f"{split}: the tissues do not share the same genes, in the same order "
            f"(gene counts {sizes}). Matched tissues are combined gene by gene."
        )

    if problems:
        raise DataError(
            f"{len(problems)} problem(s) with the {split} data:\n  - "
            + "\n  - ".join(problems)
        )

    def compare_splits(train: ad.AnnData, validation: ad.AnnData) -> SplitOverlap:
    """Measure what a training split and a validation split really have in common.

    Compares values, not just names, and counts the samples that match on both.
    """
    validation_names = set(validation.obs_names)
    shared = [name for name in train.obs_names if name in validation_names]

    n_identical = 0
    if shared:
        a = _dense(train[shared].X)
        b = _dense(validation[shared].X)
        if a.shape == b.shape:
            n_identical = int(np.all(a == b, axis=1).sum())

    return SplitOverlap(
        n_train=train.n_obs,
        n_validation=validation.n_obs,
        n_shared_names=len(shared),
        n_identical_rows=n_identical,
    )

    def warn_if_validation_repeats_training(
    train_tissues: list[ad.AnnData],
    validation_tissues: list[ad.AnnData],
) -> list[SplitOverlap]:
    """Say so, loudly, when validation is really the training data again.

    Returns one :class:`SplitOverlap` per tissue so a caller can decide what to
    do. Warns rather than raises: a cohort too small to divide has nowhere else
    to go, and refusing would only push people back to the workaround. What it
    will not do is let the situation pass unremarked, which is what version 0.1
    did on every real dataset it was run on.
    """
    overlaps = [
        compare_splits(train, validation)
        for train, validation in zip(train_tissues, validation_tissues, strict=True)
    ]

    repeated = [i for i, o in enumerate(overlaps, start=1) if o.is_effectively_identical]
    if repeated:
        which = (
            f"tissue {repeated[0]}" if len(repeated) == 1 else f"tissues {repeated}"
        )
        warnings.warn(
            f"the validation data repeats the training data for {which}: the same "
            f"samples with the same counts. Any validation loss reported will be a "
            f"training loss, and will say nothing about how the model generalises. "
            f"If the cohort is too small to divide, leave validation_path unset "
            f"instead, and no validation loss will be reported.",
            ValidationRepeatsTrainingWarning,
            stacklevel=2,
        )

    # Keyed on samples that match by value, not by name. Keying it on names
    # reported the simple vignette, which is two independent draws that happen
    # to number their samples the same way, as fully overlapping.
    partial = [
        i
        for i, o in enumerate(overlaps, start=1)
        if not o.is_effectively_identical and o.n_identical_rows > 0
    ]
    if partial:
        detail = ", ".join(
            f"tissue {i}: {overlaps[i - 1].n_identical_rows} of "
            f"{overlaps[i - 1].n_train} ({overlaps[i - 1].fraction_repeated:.0%})"
            for i in partial
        )
        warnings.warn(
            f"some training samples reappear in the validation data ({detail}). "
            f"The validation loss will be optimistic by however much they overlap.",
            ValidationRepeatsTrainingWarning,
            stacklevel=2,
        )

    return overlaps

    preffect.data.selection ¶

    Choosing which genes and which samples to use.

    Genes are chosen once, from the training data, and then applied by name. Version 0.1 chose them independently for every split: each one ranked its own genes by variability and kept its own top N. On the simple vignette, asking for the twenty most variable genes gave training and validation sets that disagreed about nine of the twenty, so column j of one matrix and column j of the other referred to different genes. The shape check compared widths, which matched, so nothing caught it. That is defect 18 in how_this_project_breaks.md.

    Applying by name rather than by position is the second half of the fix. Two files can hold the same genes in a different order, and selecting positions computed from one file would then silently take the wrong columns from another.

    def rank_genes_by_variability(adata: ad.AnnData) -> list[str]:
    """Rank genes most variable first, by interquartile range.

    The interquartile range is the spread of the middle half of a gene's counts
    across samples: the 75th percentile minus the 25th. It is used in
    preference to the variance because a single extreme count, which fRNA-seq
    data has plenty of, moves the variance a long way and the interquartile
    range hardly at all.
    """
    counts = _dense(adata.X).astype(float)
    if counts.size == 0:
        return []
    upper, lower = np.nanpercentile(counts, [75, 25], axis=0)
    spread = upper - lower
    # Ties broken by position, so the result does not depend on sort internals.
    order = np.lexsort((np.arange(len(spread)), -spread))
    return [str(adata.var_names[i]) for i in order]

    Hiding entries¶

    preffect.data.masking ¶

    Hiding counts on purpose, so that recovering them can be scored.

    Masking sets a fraction of the counts to zero before the model sees them. The model is then asked to reconstruct the whole matrix, and the entries that were hidden are the ones used to judge how well it imputes.

    "Missing completely at random" (MCAR) means the hidden entries are chosen without reference to their values or to anything about the sample: every entry of a gene is equally likely to be hidden. It is the simplest of the standard missingness patterns and the only one version 0.1 implemented, despite offering two others in its settings that raised NotImplementedError when selected.

    The fraction is applied per gene, matching version 0.1: each gene has the same proportion of its samples hidden, rather than that proportion being spread across the matrix as a whole. On a matrix with many all-zero genes the two differ, so the behaviour is kept and written down rather than quietly improved.

    Entries are hidden whether or not they were already zero. Nothing is scored as the recovery of a zero by a zero: the relative error leaves zero-truth entries out and reports how many (finding 25). What a hidden zero does is take nothing away from the model's input, so "a fifth hidden" overstates what the model lost by the zero share of the matrix: half a per cent at the project's gene selection, a fifth of the entries on an unselected matrix. Measured in exp/14 on 2026-10-29: drawing from the non-zero entries instead moves no number outside noise, so the scheme stays and the count of entries an error was computed over is reported beside it.

    Hiding entries lowers a sample's total, and the model is told by how much¶

    A sample's library size is its total count, and it sets the scale of everything the model predicts for that sample. Taken from a matrix with a fifth of its entries zeroed it is a fifth too small, while the targets keep the originals, so every predicted count came out short by the hidden fraction and the error on the hidden entries was about 2.7 times what it should have been (finding 56). :meth:Mask.library_scale is what repairs it: the factor a row's observed total is multiplied by to estimate the total it would have had, which is the number of genes divided by the number of entries of that row the model was shown.

    The correction reads how many entries were hidden, never which values. The true total and the visible entries together give the sum of the hidden ones exactly, so taking the library from the targets would hand the model the answer it is being scored on. A count of hidden entries is protocol, known before any value is looked at, and gives an estimate rather than the answer.

    It is available for "missing completely at random" and refused otherwise. The estimate is unbiased only because MCAR makes the entries the model was shown a random subsample of the row, so their mean stands in for the row's. Under a scheme that hides entries because of their values, the visible ones are not representative and the same arithmetic would be quietly wrong. Both other schemes raise NotImplementedError today, so the refusal costs nothing now and could not be added later without auditing every result produced in between.

    entry was set to zero. ``scheme`` records how they were picked, because
    the library-size correction is only valid under one of them.

    hidden: np.ndarray
    #: How the hidden entries were chosen. ``"mcar"``, missing completely at
    #: random, is the only scheme built, and the only one
    #: :meth:`library_scale` will answer for.
    scheme: str = MCAR

    @property
    def n_hidden(self) -> int:
        return int(self.hidden.sum())

    @property
    def fraction_hidden(self) -> float:
        return self.n_hidden / self.hidden.size if self.hidden.size else 0.0

    def positions(self) -> np.ndarray:
        """The hidden entries as an array of (sample, gene) pairs."""
        return np.argwhere(self.hidden)

    def visible_per_sample(self) -> np.ndarray:
        """How many of each sample's genes the model was shown, as ``[samples]``."""
        return self.hidden.shape[1] - self.hidden.sum(axis=1)

    def library_scale(self) -> np.ndarray:
        """What a sample's observed total is multiplied by to estimate its true total.

        The number of genes divided by the number of that sample's entries the
        model was shown. A sample with a fifth of its entries hidden has its
        observed total raised by about a quarter, and the exact factor differs
        from sample to sample because the hidden entries are drawn per gene
        rather than per sample.

        Returned as ``[samples, 1]`` so it multiplies a column of row totals
        directly.

        :raises DataError: for any scheme but ``"mcar"``, naming why. The
            estimate stands on the shown entries being a random subsample of
            the sample's, which only missing completely at random guarantees.
        :raises DataError: if any sample has no visible entry at all, since
            nothing is then left to estimate its total from.
        """

                f"the library-size correction under a mask is only valid for the "
                f"{MCAR!r} scheme and this mask says {self.scheme!r}. The "
                f"correction multiplies a sample's observed total by its genes "
                f"over the entries it was shown, which estimates the true total "
                f"only when the shown entries are a random subsample of the "
                f"sample's. A scheme that hides entries according to their "
                f"values breaks that, and the estimate would be biased rather "
                f"than unavailable."

                f"{empty} sample(s) have every entry hidden, so there is nothing "
                f"left to estimate their total counts from. Hide a smaller "
                f"fraction."
            )

        # the call and affected everything built afterwards.
        batch_size = n_samples

    batches = [order[i : i + batch_size] for i in range(0, n_samples, batch_size)]
    if drop_last and len(batches[-1]) < batch_size:
        batches.pop()
    return batches

    preffect.data.tensors ¶

    Turning checked count data into the tensors a model takes.

    Keeps two count matrices apart, and the distinction matters:

    • inputs are what the model sees. With masking on, some entries have been set to zero on purpose.
    • targets are what it is scored against, always the original counts.

    Version 0.1 kept both as well, under the names Xs and Rs, but chose between them inside the training loop with a test on the task name, so a run could be scored against the wrong one depending on a string.

    A third thing travels with a masked split: how much of each sample's total the hiding took away. The model scales everything it predicts for a sample by that sample's total count, and read off the inputs that total is short by the hidden fraction while the targets are not. library_scale is the factor that repairs it, derived from the mask whenever there is one, and finding 56 is what happened without it.

    targets: torch.Tensor
    corrections: torch.Tensor | None = None
    batch_labels: torch.Tensor | None = None
    mask: Mask | None = None

    #: for the simple model. An edge list rather than a square matrix because
    #: these networks are very sparse; see data/graph.py.
    edge_index: torch.Tensor | None = None
    #: What each sample's observed total has to be multiplied by to estimate
    #: the total it would have had with nothing hidden, as ``[samples, 1]``.
    #: Filled in from the mask when there is one and left None when there is
    #: not, so a split with nothing hidden is untouched. See
    #: :meth:`~preffect.data.masking.Mask.library_scale` and finding 56.
    library_scale: torch.Tensor | None = None

    def __post_init__(self) -> None:
        if self.inputs.shape != self.targets.shape:
            raise DataError(
                f"inputs are shaped {tuple(self.inputs.shape)} and targets "
                f"{tuple(self.targets.shape)}; they have to match"
            )
        for name in ("corrections", "batch_labels", "library_scale"):
            value = getattr(self, name)
            if value is not None and value.shape[0] != self.inputs.shape[0]:
                raise DataError(
                    f"{name} has {value.shape[0]} rows but there are "
                    f"{self.inputs.shape[0]} samples"
                )
        # Derived here rather than at every call site, so that an experiment
        # which builds its own TensorData with a mask gets the correction
        # without being edited. A scale supplied by hand wins, which is how
        # ``select`` carries one into a batch: a batch has no mask of its own.
        if self.mask is not None and self.library_scale is None:
            scale = torch.from_numpy(self.mask.library_scale()).to(
                device=self.inputs.device, dtype=self.inputs.dtype
            )
            if scale.shape[0] != self.inputs.shape[0]:
                raise DataError(
                    f"the mask covers {scale.shape[0]} samples but there are "
                    f"{self.inputs.shape[0]}"
                )
            object.__setattr__(self, "library_scale", scale)

        edge survives only when both endpoints are in it. Most are not; see
        data/graph.py.

        a batch is a handful of its rows. The library-size correction does, as

        f"this tissue, and a padded row carries no annotations, so a run that "
        f"pads cannot also use correction variables."

    columns = []
    for variable in variables:
        values = adata.obs[variable.name]
        if variable.kind == "categorical":

    The sample-sample network¶

    preffect.data.graph ¶

    The sample-sample network: reading it, and cutting it down to a mini-batch.

    The graph models attach a network saying which samples resemble each other. It is stored in the count file under obsm['sample_sample_adj'] as a square matrix, one row and column per sample, with a one where two samples are linked.

    Edges are kept as a list, not a square matrix¶

    A square matrix of 20,000 samples is 1.6 GB and almost entirely zeros: these networks have three or four neighbours per sample. The edge list is a few hundred kilobytes. The square form is rebuilt only for the handful of samples in a mini-batch, where it is small.

    Most edges do not survive a mini-batch, and that is reported¶

    An edge is usable only when both of its endpoints are in the same batch, and the share that survives is the batch size divided by the sample count. On a cohort of a few thousand that is a small fraction; on this project's real datasets, which run from 48 to 528 samples, it is between 38% and all of them. Version 0.1 did the same thing silently either way. Here :func:edge_survival measures it so a run can say so.

    Finding 22 in how_this_project_breaks.md; whether to batch the graph properly instead is issue #12.

    def load_adjacency(adata: ad.AnnData, split: str = "data") -> torch.Tensor:
    """Read the network as an edge list of shape ``[2, edges]``.

    Both directions of every edge are included, which is what a graph attention
    layer expects. Self-links are dropped: every sample attends to itself
    anyway, and counting them would flatter any measure of how well the network
    is reconstructed.

    :raises DataError: if the network is missing or not square.
    """
    if not has_adjacency(adata):
        raise DataError(
            f"{split} has no sample-sample network under obsm[{ADJACENCY_KEY!r}], "
            f"which the single and full models need. Use the simple model, or "
            f"add one. Present: {sorted(adata.obsm) or '(nothing)'}"
        )

    matrix = adata.obsm[ADJACENCY_KEY]
    dense = np.asarray(matrix.todense()) if hasattr(matrix, "todense") else np.asarray(matrix)

    n = adata.n_obs
    if dense.shape != (n, n):
        raise DataError(
            f"{split}'s network is shaped {dense.shape} but there are {n} samples; "
            f"it has to be square with one row per sample"
        )

    dense = np.array(dense)
    self_links = int(np.count_nonzero(np.diag(dense)))
    np.fill_diagonal(dense, 0)
    rows, cols = np.nonzero(dense)

    if rows.size == 0:
        raise DataError(
            f"{split}'s sample-sample network has no edges between different "
            f"samples"
            + (f" (only {self_links} self-links)" if self_links else "")
            + f". A graph model over {n} unconnected samples has nothing to "
            f"attend to: it costs about ten times the parameters and twenty "
            f"times the time of the simple model and cannot use any of it. "
            f"Use the simple model, or supply a network with real edges. "
            f"Every dataset version 0.1 shipped carries an identity matrix "
            f"here, so this is the common case, not a rare one. See findings "
            f"35 and 54."
        )

    return torch.from_numpy(np.vstack([rows, cols]).astype(np.int64))

    Several tissues¶

    preffect.data.tissues ¶

    Matching samples across tissues, for the full model.

    The full model takes several profiles from the same patient, for example a tumour and the normal tissue beside it, and uses them together. To do that it has to know which row of one file corresponds to which row of another.

    The files do not share sample names, deliberately. In the project's own data the names are tau1_0 and tau2_0, so a join on names would match nothing. The correspondence is written down instead, in a table under obsm['sample_association'] with one column per tissue, giving each sample's counterpart in every other tissue. A patient with no counterpart in some tissue has an empty cell in that tissue's column.

    A patient missing a tissue is padded, not refused¶

    A cohort where some patients lack a matched normal is the common case in practice, and until 2026-11-02 version 0.2 refused one outright. It now does what version 0.1 did, ported rule for rule and then measured:

    • Every empty cell becomes a row of its own in that tissue, called a ghost row here and in version 0.1. It gets a name of its own, every annotation missing, and its counts set to the rounded per-gene mean of that tissue's real samples.
    • The tissue's network is enlarged to cover the ghost rows, with no links, so a ghost attends to nothing and nothing attends to it.
    • Ghost rows are left out of the expression reconstruction term, the two library-size terms and the imputation scores, and kept in the latent and network terms. That split is version 0.1's, made in code rather than stated: it commented out the lines that would have removed ghosts from the latent and network parameters. The decision record of 2026-11-02 names it as inherited and measures it.
    • Hidden entries are never drawn from a ghost row, which :class:MultiTissueData refuses rather than trusts.

    align_tissues(..., unmatched="refuse") brings the old refusal back. That is not a historical courtesy: dropping the patients who are missing a tissue is the real alternative to padding, and it is the arm the measurement in exp/19_unmatched_samples_padded compares padding against.

    What a ghost row cannot carry¶

    Its annotations are all missing, by the rule above, so a padded run has no correction variable and no batch label for it. Correction variables are refused by name when any value is missing (:func:~preffect.data.tensors.encode_corrections), which means a padded run and correction variables cannot be combined today. Fill the annotations in yourself, or drop the incomplete patients, or leave the correction variables out.

class MultiTissueData:
    """Several tissues whose rows refer to the same patients, in the same order.

    ``tissues[0]`` is the target: the profile the full model predicts. The rest
    inform that prediction without being predicted themselves, which is what
    version 0.1 does and what the paper describes as stabilising a profile
    using matched tissues.

    ``ghosts`` records which rows were padded in for a patient missing that
    tissue, one boolean tensor per tissue. Leave it unset when the cohort is
    complete: the terms that leave ghost rows out are then skipped entirely
    rather than given an all-true selection, so a complete cohort trains bit
    for bit as it did before padding existed.
    """

    tissues: list[TensorData]
    #: One ``[samples]`` boolean tensor per tissue, True where the row is
    #: padding. None when no tissue has any, which is the ordinary case.
    ghosts: list[torch.Tensor] | None = None

    def __post_init__(self) -> None:
        if not self.tissues:
            raise DataError("a multi-tissue dataset needs at least one tissue")
        sizes = {t.n_samples for t in self.tissues}
        if len(sizes) > 1:
            raise DataError(
                f"the tissues have different sample counts {sorted(sizes)}; they "
                f"must be aligned first"
            )
        genes = {t.n_genes for t in self.tissues}
        if len(genes) > 1:
            raise DataError(f"the tissues have different gene counts {sorted(genes)}")

        if self.ghosts is None:
            return
        if len(self.ghosts) != len(self.tissues):
            raise DataError(
                f"there are {len(self.tissues)} tissue(s) but {len(self.ghosts)} "
                f"record(s) of which rows are padding; there must be one per tissue"
            )
        for index, (tissue, ghost) in enumerate(
            zip(self.tissues, self.ghosts, strict=True), start=1
        ):
            if ghost.dtype is not torch.bool or ghost.dim() != 1:
                raise DataError(
                    f"tissue {index}'s record of padded rows is {ghost.dtype} of "
                    f"shape {tuple(ghost.shape)}; it has to be one boolean per row"
                )
            if ghost.shape[0] != tissue.n_samples:
                raise DataError(
                    f"tissue {index} has {tissue.n_samples} rows but its record of "
                    f"padded rows covers {ghost.shape[0]}"
                )
            self._refuse_hidden_ghosts(index, tissue, ghost)

    @staticmethod
    def _refuse_hidden_ghosts(index: int, tissue: TensorData, ghost: torch.Tensor) -> None:
        """A hidden entry in a padded row would score the model on a value nobody measured.

        A ghost row's counts are the per-gene mean of the real samples, so
        hiding one of them and asking the model to recover it scores it against
        something this package made up. Version 0.1 kept ghost rows out of the
        masked-entry bookkeeping in its batch iterator; this refuses the mask
        instead, because a mask that hides a ghost entry was built wrongly and
        quietly dropping those entries would leave the caller's count of hidden
        entries disagreeing with what was scored.
        """
        if tissue.mask is None or not bool(ghost.any()):
            return
        rows = ghost.detach().cpu().numpy()
        hidden_in_ghosts = int(tissue.mask.hidden[rows].sum())
        if hidden_in_ghosts:
            where = "the target tissue" if index == 1 else f"tissue {index}"
            raise DataError(
                f"the mask for {where} hides {hidden_in_ghosts} entr(ies) in rows "
                f"that were padded in for patients missing that tissue. Those "
                f"counts are the per-gene mean of the real samples, so recovering "
                f"them measures nothing. Build the mask over the real rows and "
                f"leave the padded ones visible."
            )

    @property
    def n_tissues(self) -> int:
        return len(self.tissues)

    @property
    def n_samples(self) -> int:
        return self.tissues[0].n_samples

    @property
    def n_genes(self) -> int:
        return self.tissues[0].n_genes

    @property
    def target(self) -> TensorData:
        """The tissue whose counts the model predicts."""
        return self.tissues[0]

    @property
    def n_ghosts(self) -> list[int]:
        """How many rows of each tissue are padding."""
        if self.ghosts is None:
            return [0] * self.n_tissues
        return [int(g.sum()) for g in self.ghosts]

    def target_real_rows(self) -> torch.Tensor | None:
        """Which rows of the target tissue are real samples, or None if all are.

        None rather than an all-true tensor on purpose. The objective skips the
        selection entirely when this is None, so a cohort with nothing padded
        gives the same numbers to the last bit as it did before padding
        existed.
        """
        if self.ghosts is None or not bool(self.ghosts[0].any()):
            return None
        return ~self.ghosts[0]

    def select(self, positions: np.ndarray | torch.Tensor) -> MultiTissueData:
        """One mini-batch, taken from every tissue at the same positions.

        The same positions in every tissue, which is only meaningful because
        the rows were aligned first. The record of which rows are padding is
        taken at the same positions, so a batch knows which of its own rows
        are padding without having to look back at the whole split.
        """
        tissues = [t.select(positions) for t in self.tissues]
        if self.ghosts is None:
            return MultiTissueData(tissues)
        ghosts = [
            g[torch.as_tensor(positions, dtype=torch.long, device=g.device)]
            for g in self.ghosts
        ]
        return MultiTissueData(tissues, ghosts=ghosts)

    def to(self, device: str | torch.device) -> MultiTissueData:
        """Return a copy with every tissue's tensors moved to ``device``.

        Every tissue goes to the same device. The full model's terms are
        summed across tissues, so a split set would fail at the first sum, and
        failing here with a clear message beats failing there with an obscure
        one. The record of which rows are padding moves too: it selects rows of
        tensors that have moved.
        """
        return MultiTissueData(
            [t.to(device) for t in self.tissues],
            ghosts=None if self.ghosts is None else [g.to(device) for g in self.ghosts],
        )

            f"default in place, which pads them."

        filled[column] = names
        _check_column(column, [n for n, g in zip(names, ghost, strict=True) if not g],
                      adata, index, split)
        ghosts.append(ghost)
        plans.append((adata, names, ghost))

    # Built in a second pass so that every column of the table carries its
    # padded names before any tissue stores a copy of it.
    tissues = [_pad_tissue(adata, names, ghost, filled) for adata, names, ghost in plans]
    return TissueAlignment(tissues=tissues, ghosts=ghosts)

    Example data¶

    preffect.example_data ¶

    A small simulated cohort to try PREFFECT on, written by preffect example-data.

    Three splits with no sample in common, for each of the three models: the simple model, the single model (the same counts plus a sample-sample network with real edges in it) and the full model (two tissues per patient, under different names, with the correspondence written down). About two seconds, 200 genes, three groups of samples that each raise a different block of genes, so there is something real to find and you can tell whether the model found it. The counts are negative binomial, which is what PREFFECT models.

    Version 0.1's vignettes ship a test/ folder byte-identical to validation/, in all three of simple, single and full, and a published figure shows a user pointing inference at that folder while the caption calls it a held-out test set. The data here is generated rather than committed, for two reasons: counts do not belong in a git repository, and a generator can promise what a committed file cannot. The three splits are drawn from disjoint samples by construction, and :func:write_example_data checks that and raises rather than write them if they are not. Issue #11.

    The full model's two files per split deliberately do not share sample names, as the project's own data does not, so the correspondence goes in a table under obsm['sample_association'] with one column per tissue, which is where :func:preffect.data.tissues.align_tissues looks. unmatched takes the second tissue away from a share of patients, which is the shape of a real cohort where some patients have no matched normal, and is what the full model's padding exists for. It is zero by default and draws no random numbers when it is zero, so the files written without it are byte for byte the files written before the option existed.

    def simulate(
    n_samples: int,
    n_genes: int,
    rng: np.random.Generator,
    offset: int,
    groups: np.ndarray | None = None,
    lift: float = 4.0,
):
    """Counts with a subtype signal in them, so an example has something to find.

    Each sample belongs to one of three groups, and each group raises a
    different block of genes. Negative binomial, which is what PREFFECT models,
    so the example is not quietly fitting Gaussian data.

    :param n_samples: rows to draw.
    :param n_genes: columns to draw.
    :param rng: the generator; every draw comes from it, so a seed fixes the file.
    :param offset: where sample numbering starts, so no name repeats across splits.
    :param groups: the group each sample belongs to. Drawn here when left
        unset; passed in for the full model's second tissue, which is the same
        patients and so must carry the same groups.
    :param lift: how much a group raises its own block of genes. Lower for a
        second tissue, which stands for the quieter normal tissue beside a
        tumour and should not simply repeat the first.
    :returns: the counts, the group of each sample, and the sample names.
    """
    if groups is None:
        groups = rng.integers(0, N_SUBTYPES, size=n_samples)
    base = rng.lognormal(mean=3.0, sigma=1.0, size=n_genes)

    means = np.tile(base, (n_samples, 1))
    block = n_genes // (N_SUBTYPES + 1)
    for group in range(N_SUBTYPES):
        rows = groups == group
        means[np.ix_(rows, np.arange(group * block, (group + 1) * block))] *= lift

    depth = rng.lognormal(mean=0.0, sigma=0.25, size=(n_samples, 1))
    means = means * depth

    dispersion = rng.uniform(2.0, 8.0, size=n_genes)
    probability = dispersion / (dispersion + means)
    counts = rng.negative_binomial(dispersion, probability).astype(np.float32)

    names = [f"SAMPLE_{offset + i:04d}" for i in range(n_samples)]
    return counts, groups, names

    def ring_network(n: int) -> np.ndarray:
    """A sample-sample network with real edges, for the single-model example.

    Not an identity matrix. Every real dataset version 0.1 shipped carries one
    of those, which is a network with nothing in it, and a graph model on it
    has nothing to attend to. Findings 35 and 54.

    network = np.zeros((n, n), dtype=np.int64)
    for i in range(n):
        network[i, (i + 1) % n] = network[(i + 1) % n, i] = 1
        network[i, (i + 5) % n] = network[(i + 5) % n, i] = 1
    return network

    def write_example_data(out: Path, *, seed: int = 0, unmatched: float = 0.0) -> ExampleData:
    """Write the example cohort under ``out`` and check the splits are distinct.

    Writes ``<out>/{simple,single}/{train,validation,test}/counts.tau_1.h5ad``
    and, for the full model, two tissues per split,
    ``<out>/full/{train,validation,test}/counts.tau_{1,2}.h5ad``. The same seed
    writes the same files.

    :param out: the folder to write under; created if missing.
    :param seed: seed for every draw.
    :param unmatched: the share of patients whose second tissue is left out of
        the full model's data, at least 0 and below 1. The first tissue is never
        thinned, because it is the one the model predicts.
    :raises ConfigError: if ``unmatched`` is outside its range.
    :raises DataError: if any two splits share a sample or are the same file,
        which is the promise this exists to keep.
    :returns: what was written.
    """
    import anndata as ad
    import pandas as pd

    if not 0.0 <= unmatched < 1.0:
        raise ConfigError(f"unmatched must be at least 0 and below 1, got {unmatched}")

    out = Path(out)
    rng = np.random.default_rng(seed)
    offset = 0
    written: dict[str, dict[str, Path]] = {}
    sample_names: dict[str, list[str]] = {}

    for kind in ("simple", "single"):
        written[kind] = {}
        for split, n in N_SAMPLES.items():
            counts, groups, names = simulate(n, N_GENES, rng, offset)
            offset = _advance(offset, n)  # so no sample name is ever reused across splits

            adata = ad.AnnData(counts)
            # Object dtype, as the package writes names: under pandas 3 a list
            # becomes a string array that anndata 0.11 refuses to write, and a
            # fresh install can resolve exactly that pairing. Finding 48.
            adata.obs_names = pd.Index(names, dtype=object)
            adata.var_names = pd.Index([f"GENE_{j:04d}" for j in range(N_GENES)], dtype=object)
            adata.obs["subtype"] = pd.Categorical([f"group_{g}" for g in groups])
            adata.obs["batch"] = pd.Categorical(["A" if i % 2 else "B" for i in range(n)])
            if kind == "single":
                adata.obsm["sample_sample_adj"] = ring_network(n)

            folder = out / kind / split
            folder.mkdir(parents=True, exist_ok=True)
            # The loader looks for '*tau_<n>.h5ad', where n is the tissue
            # number. An example with one tissue is tau_1.
            path = folder / "counts.tau_1.h5ad"
            adata.write_h5ad(path)
            written[kind][split] = path
            sample_names[f"{kind}/{split}"] = names

    # The full model: two tissues per split, the same patients in each, and a
    # table saying which row of one is which row of the other. Written after
    # the others so the sample names carry on rather than repeating theirs.
    written["full"] = {}
    full_tissue_files: dict[str, list[Path]] = {}
    unmatched_written: dict[str, int] = {}
    for split, n in N_SAMPLES.items():
        first_counts, groups, first_names = simulate(n, N_GENES, rng, offset)
        offset = _advance(offset, n)
        # The same patients, so the same groups; a weaker block lift, so the
        # second tissue informs the first rather than repeating it.
        second_counts, _, second_names = simulate(n, N_GENES, rng, offset, groups=groups, lift=1.8)
        offset = _advance(offset, n)

        # Which patients keep their second tissue. Every one of them unless
        # asked otherwise, and the draw is skipped entirely at zero so that the
        # generator's stream, and so every file, is untouched.
        kept = [np.ones(n, dtype=bool), np.ones(n, dtype=bool)]
        n_drop = int(round(unmatched * n))
        if n_drop:
            kept[1][rng.choice(n, size=n_drop, replace=False)] = False
        unmatched_written[split] = int((~kept[1]).sum())

        # One column per tissue, each listing that tissue's name for the
        # patient on this row. The two tissues share no sample name, as the
        # project's own data does not, so this table is the only link. A
        # patient whose second tissue was taken away has an empty cell there,
        # which is how a real cohort records a missing matched normal.
        association = pd.DataFrame({"tau_1": first_names, "tau_2": second_names})
        # The empty cell is written as the string "NA", which version 0.1 used
        # and which the reader turns into a missing value, rather than as NaN:
        # under pandas 2 a column of names with a float NaN in it is an object
        # column that anndata refuses to write ("non-string objects"), while
        # pandas 3 makes it a string column and writes it. Continuous integration
        # on Python 3.11 resolved pandas 2 and failed on exactly this line.
        association.loc[~kept[1], "tau_2"] = "NA"

        folder = out / "full" / split
        folder.mkdir(parents=True, exist_ok=True)
        paths = []
        written_names = []
        for tissue, (tissue_counts, tissue_names, rows) in enumerate(
            ((first_counts, first_names, kept[0]), (second_counts, second_names, kept[1])),
            start=1,
        ):
            own_names = [name for name, keep in zip(tissue_names, rows, strict=True) if keep]
            own_groups = groups[rows]
            m = len(own_names)
            adata = ad.AnnData(tissue_counts if rows.all() else tissue_counts[rows])
            adata.obs_names = pd.Index(own_names, dtype=object)
            adata.var_names = pd.Index([f"GENE_{j:04d}" for j in range(N_GENES)], dtype=object)
            adata.obs["subtype"] = pd.Categorical([f"group_{g}" for g in own_groups])
            adata.obs["batch"] = pd.Categorical(["A" if i % 2 else "B" for i in range(m)])
            adata.obsm["sample_sample_adj"] = ring_network(m)
            # An obsm frame is aligned by index, so each tissue carries the
            # table under its own names, holding the rows for the patients it
            # actually has. On a complete cohort that is the whole table in
            # both files.
            own = association[rows].copy()
            own.index = pd.Index(own_names, dtype=object)
            adata.obsm["sample_association"] = own
            path = folder / f"counts.tau_{tissue}.h5ad"
            adata.write_h5ad(path)
            paths.append(path)
            written_names.extend(own_names)

        full_tissue_files[split] = paths
        written["full"][split] = paths[0]
        sample_names[f"full/{split}"] = written_names

    # The promise this exists to keep. Checked before reporting success,
    # because version 0.1's vignettes failed exactly here and nothing noticed.
    problems = []
    for kind in KINDS:
        splits = list(N_SAMPLES)
        for i, a in enumerate(splits):
            for b in splits[i + 1 :]:
                shared = set(sample_names[f"{kind}/{a}"]) & set(sample_names[f"{kind}/{b}"])
                if shared:
                    problems.append(f"{kind}: {a} and {b} share {len(shared)} sample(s)")
                digest_a = hashlib.md5(written[kind][a].read_bytes()).hexdigest()
                digest_b = hashlib.md5(written[kind][b].read_bytes()).hexdigest()
                if digest_a == digest_b:
                    problems.append(f"{kind}: {a} and {b} are the same file")
    # The second tissue has to differ from the first as well, or the full
    # model would be combining a profile with a copy of itself.
    for split, paths in full_tissue_files.items():
        digests = [hashlib.md5(p.read_bytes()).hexdigest() for p in paths]
        if len(set(digests)) != len(digests):
            problems.append(f"full: {split}'s two tissues are the same file")

    if problems:
        raise DataError("the splits are not distinct: " + "; ".join(problems))

    return ExampleData(out=out, files=written, unmatched=unmatched_written)
