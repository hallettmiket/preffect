    The three models¶

    simple uses counts alone. single adds one sample-sample network. full takes several matched tissues.

    Asking for one and getting another is not possible: a request that cannot be honoured raises rather than quietly becoming a different model.

    Layers¶

    preffect.layers ¶

    The network pieces the simple model is built from.

    Four of them: an encoder and decoder for the counts, and a smaller pair for the library size. The graph versions, which replace the count encoder for the single and full models, arrive with those models.

    Each class builds only what it uses. Version 0.1 had one encoder class that constructed the graph path and the plain path together and then chose between them at call time. For a simple model on twenty thousand genes that allocated a graph attention layer of sixteen million parameters which was never touched: measured at 94% of the encoder's weights, 64 MB, initialised on every construction and written into every saved session. Separate classes cost a little duplication and remove all of that.

    Terms used here¶
    • Encoder maps counts to a small set of numbers, the latent space, that summarise a sample. Decoder maps those numbers back out.
    • Latent space has a mean and a variance per dimension, because the model is variational: it learns a distribution over the summary rather than a single value.
    • Correction variables are things like sequencing batch that the model is told about so it can account for them.
    • FiLM, feature-wise linear modulation, lets a correction variable scale and shift every feature rather than merely being pasted on as extra columns.

    class FiLM(nn.Module):
    """Scale and shift every feature, by an amount the conditioning decides.

    Produces one multiplier and one offset per feature from the conditioning
    values, so a correction variable can modulate the whole representation
    instead of being concatenated to it.
    """

    def __init__(self, n_conditions: int, n_features: int):
        super().__init__()
        self.to_scale_and_shift = nn.Linear(n_conditions, 2 * n_features)

    def forward(self, features: torch.Tensor, conditions: torch.Tensor) -> torch.Tensor:
        scale, shift = self.to_scale_and_shift(conditions.float()).chunk(2, dim=-1)
        return scale * features + shift

    class CountEncoder(nn.Module):
    """Counts to a latent mean and log variance, for the simple model.

    Correction variables are appended to the input and also drive two FiLM
    layers. Appending them and modulating with them is version 0.1's
    arrangement and is kept.
    """

    def __init__(
        self,
        n_genes: int,
        n_corrections: int,
        hidden_dim: int,
        latent_dim: int,
        leaky_relu_slope: float = 0.2,
        dropout: float = 0.1,
    ):
        super().__init__()
        self.n_corrections = n_corrections
        input_dim = n_genes + n_corrections
        wide = hidden_dim + n_corrections

        self.layer1 = nn.Linear(input_dim, wide)
        self.layer2 = nn.Linear(wide, hidden_dim)
        self.to_mean = nn.Linear(hidden_dim, latent_dim)
        self.to_log_variance = nn.Linear(hidden_dim, latent_dim)

        self.film1 = FiLM(n_corrections, wide) if n_corrections else None
        self.film2 = FiLM(n_corrections, hidden_dim) if n_corrections else None

        self.activation = nn.LeakyReLU(negative_slope=leaky_relu_slope)
        self.dropout1 = nn.Dropout(dropout)
        self.dropout2 = nn.Dropout(dropout)

        self.apply(_orthogonal_init)

    def scale_input(self, counts: torch.Tensor) -> torch.Tensor:
        """The scale the first layer sees. See :func:`scale_counts`; a seam for ``exp/10``."""
        return scale_counts(counts)

    def combine_with_corrections(
        self, counts: torch.Tensor, corrections: torch.Tensor | None
    ) -> torch.Tensor:
        """Put the correction variables together with the counts.

        A seam, not a setting. Version 0.1 concatenates their raw values here
        while embedding the categorical ones in the library encoder, with no
        explanation for the difference. Experiment ``exp/02`` overrides this to
        try the embedding treatment in both. See
        ``decisions/2026-10-15_odd_choices_kept_pending_measurement.md``.
        """
        if not self.n_corrections:
            return counts
        if corrections is None:
            raise ValueError(
                f"this encoder was built for {self.n_corrections} correction "
                f"variable(s) but none were supplied"
            )
        return torch.cat((counts, corrections.float()), dim=1)

    def project_to_latent(
        self, hidden: torch.Tensor
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """Hidden representation to a latent mean and log variance.

        No activation, unlike version 0.1, which applied a leaky rectifier to
        both. ``exp/01`` measured that across eight seeds and found no
        detectable difference in the loss, the reconstruction or the KL
        divergence term, so the simpler and more conventional form was kept.
        See ``decisions/2026-10-15_exp01_latent_rectifier_removed.md``, which
        also records the limit of that evidence: it was measured at 150 genes
        and wants repeating at realistic scale.

        Still a seam, so that repeat costs an hour rather than a day.
        """
        return self.to_mean(hidden), self.to_log_variance(hidden)

    def forward(
        self, counts: torch.Tensor, corrections: torch.Tensor | None = None
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """:returns: the latent mean and log variance, each ``[samples, latent_dim]``."""
        h = self.combine_with_corrections(self.scale_input(counts), corrections)

        h = self.layer1(h)
        h = torch.nn.functional.elu(h)
        if self.film1 is not None:
            h = self.film1(h, corrections)
        h = self.dropout1(h)

        h = self.layer2(h)
        h = torch.nn.functional.elu(h)
        if self.film2 is not None:
            h = self.film2(h, corrections)
        h = self.dropout2(h)

        return self.project_to_latent(h)

    class CountDecoder(nn.Module):
    """Latent values back to a hidden representation, for the simple model.

    Stops at the hidden layer. The model turns that into distribution
    parameters, because the three parameters share this representation.
    """

    def __init__(
        self,
        latent_dim: int,
        n_corrections: int,
        hidden_dim: int,
        leaky_relu_slope: float = 0.2,
        dropout: float = 0.1,
    ):
        super().__init__()
        self.n_corrections = n_corrections
        self.layer = nn.Linear(latent_dim + n_corrections, hidden_dim)
        self.film = FiLM(n_corrections, hidden_dim) if n_corrections else None
        self.activation = nn.LeakyReLU(negative_slope=leaky_relu_slope)
        self.dropout = nn.Dropout(dropout)
        self.apply(_orthogonal_init)

    def forward(
        self, latent: torch.Tensor, corrections: torch.Tensor | None = None
    ) -> torch.Tensor:
        """:returns: ``[samples, hidden_dim]``."""
        if self.n_corrections:
            if corrections is None:
                raise ValueError(
                    f"this decoder was built for {self.n_corrections} correction "
                    f"variable(s) but none were supplied"
                )
            h = torch.cat((latent, corrections.float()), dim=1)
        else:
            h = latent

        h = self.activation(self.layer(h))
        if self.film is not None:
            h = self.film(h, corrections)
        return self.dropout(h)

    class LibrarySizeEncoder(nn.Module):
    """A sample's total count to its own small latent space.

    Library size is the total number of reads for a sample, and it varies for
    reasons that have nothing to do with biology. Modelling it separately lets
    the rest of the model work on composition instead of absolute depth.

    The input is the log of the total, because library sizes span orders of
    magnitude.
    """

    def __init__(
        self,
        n_corrections: int,
        hidden_dim: int,
        latent_dim: int,
        leaky_relu_slope: float = 0.2,
        dropout: float = 0.1,
    ):
        super().__init__()
        self.n_corrections = n_corrections
        self.layer = nn.Linear(1 + n_corrections, hidden_dim)
        self.to_mean = nn.Linear(hidden_dim, latent_dim)
        self.to_log_variance = nn.Linear(hidden_dim, latent_dim)
        self.film = FiLM(n_corrections, hidden_dim) if n_corrections else None
        self.activation = nn.LeakyReLU(negative_slope=leaky_relu_slope)
        self.dropout = nn.Dropout(dropout)
        self.apply(_orthogonal_init)

    def forward(
        self, log_library_size: torch.Tensor, corrections: torch.Tensor | None = None
    ) -> tuple[torch.Tensor, torch.Tensor]:
        h = log_library_size.reshape(-1, 1)
        if self.n_corrections:
            if corrections is None:
                raise ValueError("correction variables were expected but not supplied")
            h = torch.cat((h, corrections.float()), dim=1)

        h = self.activation(self.layer(h))
        if self.film is not None:
            h = self.film(h, corrections)
        h = self.dropout(h)
        return self.to_mean(h), self.to_log_variance(h)

    scale_counts(counts: Tensor) -> torch.Tensor

    What the expression encoder is fed: log(1 + counts).

    Version 0.1 fed raw counts, which reach 2.4 million in the DCIS cohort. A learning-rate-sized change to one first-layer weight times an input that size moved the encoder's output a thousandfold in a single step, which was the loss excursion of issue #22 (finding 49). On the log scale the same step moves it by a few thousandths, and the spikes are gone in six seeds of six (exp/10, decision 2026-10-28), with imputation error halved and clustering recovery doubled. The targets, the decoder and the objective still see raw counts; only the encoder's first layer sees this.

    A function, not a setting, so there is one behaviour. exp/10's raw arm overrides the encoders' scale_input to feed counts unchanged.

    The simple model¶

    preffect.model ¶

    The simple model: counts in, a fitted count distribution out.

    The simple model uses the counts alone, with optional correction variables such as sequencing batch. The single and full models add a sample-sample network and matched tissues, and arrive later.

    What it does, in order¶
    1. The encoder turns a sample's counts into a mean and a variance over a small latent space, a handful of numbers summarising that sample.
    2. Reparameterising draws a point from that distribution in a way that gradients can pass through, so the model can be trained.
    3. The decoder turns the drawn point back into a hidden representation.
    4. Three heads turn that representation into the parameters of a count distribution per gene: the composition, the dispersion, and, for the zero-inflated form, the extra-zero log-odds.
    5. Separately, a library size is either taken from the data or predicted by its own small encoder and decoder. Multiplying the composition by it gives the expected count.

    The composition comes out of a softmax, so a sample's predicted proportions sum to one across genes and the library size sets the scale. That split is what lets the model describe what a sample is made of separately from how deeply it was sequenced.

    Because the library sets the scale and the composition cannot change it, a library read off a matrix with entries hidden makes every predicted count short by the hidden fraction, and no amount of training can make it up. forward therefore takes a library_scale: the factor that estimates a sample's real total back from how many of its entries were hidden. It is None when nothing is hidden, and the model is then exactly what it was. Finding 56.

    Output¶

    :class:ModelOutput holds every intermediate quantity as a named field. Version 0.1 returned nested dictionaries of lists keyed by strings, so a consumer reached for outputs['latent_spaces']['mu_simple'][0] and a typo produced a KeyError at best.

    class SimpleModel(nn.Module):
    """The simple PREFFECT model.

    :param n_genes: number of genes, fixed once the data is read.
    :param config: the model section of a checked configuration.
    """

    def __init__(self, n_genes: int, config: ModelConfig):
        super().__init__()
        self.n_genes = n_genes
        self.config = config

        n_corrections = len(config.correction_variables)

        self.encoder = CountEncoder(
            n_genes=n_genes,
            n_corrections=n_corrections,
            hidden_dim=config.hidden_dim,
            latent_dim=config.latent_dim,
            leaky_relu_slope=config.leaky_relu_slope,
            dropout=config.dropout,
        )
        self.decoder = CountDecoder(
            latent_dim=config.latent_dim,
            n_corrections=n_corrections,
            hidden_dim=config.hidden_dim,
            leaky_relu_slope=config.leaky_relu_slope,
            dropout=config.dropout,
        )

        if config.infer_library_size:
            self.library_encoder = LibrarySizeEncoder(
                n_corrections=n_corrections,
                hidden_dim=config.hidden_dim,
                latent_dim=config.latent_dim,
                leaky_relu_slope=config.leaky_relu_slope,
                dropout=config.dropout,
            )
            self.library_decoder = LibrarySizeDecoder(
                latent_dim=config.latent_dim,
                n_corrections=n_corrections,
                hidden_dim=config.hidden_dim,
                leaky_relu_slope=config.leaky_relu_slope,
            )
        else:
            self.library_encoder = None
            self.library_decoder = None

        self.to_composition = nn.Linear(config.hidden_dim, n_genes)
        self.to_dispersion = nn.Linear(config.hidden_dim, n_genes)
        self.to_zero_logits = (
            nn.Linear(config.hidden_dim, n_genes)
            if config.likelihood is Likelihood.ZINB
            else None
        )

        self.activation = nn.LeakyReLU(negative_slope=config.leaky_relu_slope)

    # -- seams -------------------------------------------------------------

    def activate_decoded(self, hidden: torch.Tensor) -> torch.Tensor:
        """Shape the decoder's output before the parameter heads read it.

        Nothing, deliberately. The decoder already ends with a leaky rectifier,
        and version 0.1 applied a second one here that nothing in the code or
        the paper explains. It is removed because one unexplained non-linearity
        fewer is simpler and the reconstruction is unaffected.

        **That is a judgement about simplicity, not a measured improvement,**
        and two earlier versions of this docstring claimed more. ``exp/03`` does
        find the same reconstruction for less than half the KL divergence
        (305.7 against 143.4 across eight seeds), but the KL divergence is not a
        property of the model alone: rescaling the latent and compensating in
        the decoder's first layer leaves the model's output identical and moves
        the KL by tens of per cent. See finding 30. A claim that the lower-KL
        latent space clusters better was also withdrawn, untested.

        Still a seam, so the comparison can be repeated at full scale, and the
        repeat should measure something functional rather than the KL. See
        ``decisions/2026-10-15_exp03_second_rectifier_removed.md``.
        """
        return hidden

    # -- the pass ----------------------------------------------------------

    def reparameterise(
        self, mean: torch.Tensor, log_variance: torch.Tensor
    ) -> torch.Tensor:
        """Draw from a Gaussian so that gradients still reach the parameters.

        Sampling directly would break the chain, because a draw is not a
        differentiable function of the distribution it came from. Drawing
        standard noise and scaling it keeps the randomness outside the path the
        gradient takes.
        """
        standard_deviation = torch.exp(0.5 * log_variance)
        return mean + torch.randn_like(standard_deviation) * standard_deviation

    def forward(
        self,
        counts: torch.Tensor,
        corrections: torch.Tensor | None = None,
        library_scale: torch.Tensor | None = None,
    ) -> ModelOutput:
        """Run one batch through the model.

        :param counts: ``[samples, genes]``, raw counts.
        :param corrections: ``[samples, correction variables]``, or None.
        :param library_scale: ``[samples, 1]``, what each sample's observed
            total is multiplied by to estimate the total it would have had
            with nothing hidden. Supplied when entries have been masked, and
            None otherwise, in which case nothing changes. Finding 56.
        """
        if counts.shape[1] != self.n_genes:
            raise ValueError(
                f"this model was built for {self.n_genes} genes but was given "
                f"{counts.shape[1]}"
            )

        cfg = self.config

        latent_mean, latent_log_variance = self.encoder(counts, corrections)
        latent_mean = latent_mean.clamp(-cfg.max_latent_mean, cfg.max_latent_mean)
        latent_log_variance = latent_log_variance.clamp(
            -cfg.max_log_variance, cfg.max_log_variance
        )
        latent = self.reparameterise(latent_mean, latent_log_variance)

        return self._finish_counts(
            counts,
            corrections,
            latent,
            latent_mean,
            latent_log_variance,
            library_scale=library_scale,
        )

    def _finish_counts(
        self,
        counts: torch.Tensor,
        corrections: torch.Tensor | None,
        latent: torch.Tensor,
        mean: torch.Tensor,
        log_variance: torch.Tensor,
        library_scale: torch.Tensor | None = None,
    ) -> ModelOutput:
        """Everything downstream of the latent values.

        Shared by all three models, and called again at inference time with the
        latent mean in place of a draw, so that a cleaned count matrix is
        reproducible.
        """
        library = self._library_size(counts, corrections, library_scale)
        hidden = self.activate_decoded(self.decoder(latent, corrections))

        composition = torch.softmax(self.to_composition(hidden), dim=1)
        dispersion = self._dispersion(hidden)
        zero_logits = (
            self.to_zero_logits(hidden) if self.to_zero_logits is not None else None
        )

        return ModelOutput(
            latent_mean=mean,
            latent_log_variance=log_variance,
            latent=latent,
            composition=composition,
            dispersion=dispersion,
            library_size=library["library_size"],
            counts=self._fitted_counts(composition, dispersion, zero_logits, library),
            zero_logits=zero_logits,
            library_latent_mean=library["latent_mean"],
            library_latent_log_variance=library["latent_log_variance"],
            library_latent=library["latent"],
            predicted_log_library_size=library["predicted_log"],
        )

    # -- pieces ------------------------------------------------------------

    def _fitted_counts(
        self,
        composition: torch.Tensor,
        dispersion: torch.Tensor,
        zero_logits: torch.Tensor | None,
        library: dict[str, torch.Tensor | None],
    ):
        """The count distribution the composition and library size imply."""
        mean_counts = composition * library["library_size"]
        if self.config.likelihood is Likelihood.ZINB:
            return ZeroInflatedNegativeBinomial(
                mu=mean_counts, theta=dispersion, zero_logits=zero_logits
            )
        return NegativeBinomial(mu=mean_counts, theta=dispersion)

    def _dispersion(self, hidden: torch.Tensor) -> torch.Tensor:
        """Positive inverse dispersion per gene per sample."""
        raw = self.to_dispersion(hidden)
        if not self.config.log_scale_dispersion:
            return torch.nn.functional.softplus(raw)
        limit = self.config.max_log_library_size
        return torch.exp(raw.clamp(-limit, limit))

    def _library_size(
        self,
        counts: torch.Tensor,
        corrections: torch.Tensor | None,
        library_scale: torch.Tensor | None = None,
    ) -> dict[str, torch.Tensor | None]:
        """Either the observed total per sample, or one the model predicts.

        The counts this reads are what the model was shown, which under a mask
        is short of the sample's real total by whatever the hiding took away.
        ``library_scale`` is the factor that estimates the real total back,
        from the **number** of entries hidden in each sample and never from
        their values. Without it the composition, which sums to one across
        genes, is multiplied by a total a fifth too small at a fifth hidden,
        and nothing in the model can recover the difference. Finding 56.

        The correction applies to both paths, because both start from the
        observed total: directly when the library is read from the data, and
        as the log total handed to the library encoder when it is inferred.
        """
        observed = counts.sum(dim=1, keepdim=True)
        if library_scale is not None:
            observed = observed * library_scale

        if self.library_encoder is None:
            return {
                "library_size": observed,
                "latent_mean": None,
                "latent_log_variance": None,
                "latent": None,
                "predicted_log": None,
            }

        cfg = self.config
        log_observed = torch.log(observed.squeeze(1) + 1)
        mean, log_variance = self.library_encoder(log_observed, corrections)
        log_variance = log_variance.clamp(-cfg.max_log_variance, cfg.max_log_variance)
        latent = self.reparameterise(mean, log_variance)

        predicted_log = self.library_decoder(latent, corrections)
        limit = cfg.max_log_library_size
        return {
            "library_size": torch.exp(predicted_log.clamp(-limit, limit)),
            "latent_mean": mean,
            "latent_log_variance": log_variance,
            "latent": latent,
            "predicted_log": predicted_log,
        }

    Graph layers¶

    preffect.graph_layers ¶

    Network pieces that use the sample-sample network.

    The single and full models attach a network saying which samples resemble each other, and let a graph attention layer pass information along its edges. A sample's representation is then built partly from its neighbours, which is meant to stabilise profiles that are individually noisy.

    Requires the optional graph extra::

    uv pip install -e ".[graph]"
    Two things about these networks, both measured¶

    They are very sparse: three or four neighbours per sample across every dataset in this project. And version 0.1 induces the subgraph on each mini-batch, so an edge survives only when both of its endpoints land in the same batch, which means the share of the network visible in an epoch is the batch size divided by the sample count.

    That share only gets small on a large cohort. This project's real datasets run from 48 to 528 samples, so at version 0.1's default batch of 200 they see between 38% and all of their network each epoch. An earlier version of this paragraph said "about a tenth at the published settings", which came from a synthetic 2,000-sample dataset used in one scratch notebook. Finding 22 in how_this_project_breaks.md, issue #12.

    The adjacency decoder is not a faithful port, and could not be¶

    Version 0.1 reconstructs the network with a linear layer whose output width is the mini-batch size, so the model's parameter count follows a training setting: 7,400 parameters at a batch of 50 and 15,050 at a batch of 200. It can only ever process batches of exactly the size it was built for, which is why version 0.1 had to discard the last short batch of every epoch. Keeping every sample was decided and tested earlier, and the two are incompatible.

    So this uses the conventional graph autoencoder decoder instead: project the latent values, then take the inner product between every pair and squash it to a probability. An edge is predicted where two samples have similar latent representations, which is what an edge is supposed to mean. It works at any batch size and has fewer parameters. Recorded as finding 23 and in results_affecting_changes.md.

        latent_dim: int,
        attention_heads: int = 8,
        leaky_relu_slope: float = 0.2,

        # Version 0.1 calls this layer3_test. It is in the live path, not a
        # leftover, so it is ported; the name is not.
        self.refine = nn.Linear(hidden_dim, hidden_dim)

        self.to_mean = nn.Linear(hidden_dim + n_corrections, latent_dim)
        self.to_log_variance = nn.Linear(hidden_dim + n_corrections, latent_dim)

        self.film1 = FiLM(n_corrections, wide) if n_corrections else None
        self.film2 = FiLM(n_corrections, hidden_dim) if n_corrections else None

        self.dropout1 = nn.Dropout(dropout)
        self.dropout2 = nn.Dropout(dropout)
        self.dropout3 = nn.Dropout(dropout)
        self.activation = nn.LeakyReLU(negative_slope=leaky_relu_slope)

        self.apply(_orthogonal_init)

    def scale_input(self, counts: torch.Tensor) -> torch.Tensor:
        """The scale the attention layer sees: ``log(1 + counts)``, as the simple model's encoder.

        Measured on the simple model (``exp/10``); the graph models get it on
        the same reasoning without a measurement of their own, which the
        decision record says and an issue tracks.
        """
        return scale_counts(counts)

    def combine_with_corrections(
        self, counts: torch.Tensor, corrections: torch.Tensor | None
    ) -> torch.Tensor:
        """Attach the correction variables to the counts, if there are any.

        Correction variables are things recorded about a sample that are not
        expression and that the model should account for, such as which batch
        it was sequenced in. They are concatenated along the feature axis, so
        the encoder sees one wider row per sample. `exp/02` tested embedding
        them instead and found no benefit, which is why this is still a
        concatenation.

        :raises ValueError: if the encoder was built expecting correction
            variables and none were supplied. Version 0.1 carried on with
            whatever happened to be there, which is the habit this package
            exists to drop.
        """
        if not self.n_corrections:
            return counts
        if corrections is None:
            raise ValueError(
                f"this encoder was built for {self.n_corrections} correction "
                f"variable(s) but none were supplied"
            )
        return torch.cat((counts, corrections.float()), dim=1)

    def project_to_latent(
        self, hidden: torch.Tensor, corrections: torch.Tensor | None
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """No activation, matching the simple model after ``exp/01``.

        ``exp/01`` tested this on the simple model only. Applying the same
        answer here is a consistency judgement rather than a measurement, and
        issue #4 covers repeating it.
        """
        if self.n_corrections and corrections is not None:
            hidden = torch.cat((hidden, corrections.float()), dim=1)
        return self.to_mean(hidden), self.to_log_variance(hidden)

    def forward(
        self,
        counts: torch.Tensor,
        edge_index: torch.Tensor,
        corrections: torch.Tensor | None = None,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """:param edge_index: ``[2, edges]``, the network for this batch."""
        h = self.combine_with_corrections(self.scale_input(counts), corrections)

        h = torch.nn.functional.elu(self.attention(h, edge_index))
        if self.film1 is not None:
            h = self.film1(h, corrections)
        h = self.dropout1(h)

        h = torch.nn.functional.elu(self.narrow(h))
        if self.film2 is not None:
            h = self.film2(h, corrections)
        h = self.dropout2(h)

        h = self.dropout3(torch.nn.functional.elu(self.refine(h)))
        return self.project_to_latent(h, corrections)

    class AdjacencyDecoder(nn.Module):
    """Latent values back to a predicted sample-sample network.

    Projects the latent values, then scores every pair by inner product and
    squashes to a probability, so an edge is predicted where two samples sit
    close together in the latent space. That is what an edge means, and it is
    the conventional graph autoencoder decoder.

    Deliberately not version 0.1's arrangement, which used a linear layer of
    width equal to the mini-batch size. See this module's docstring.
    """

    def __init__(
        self,
        latent_dim: int,
        n_corrections: int,
        hidden_dim: int,
        leaky_relu_slope: float = 0.2,
        dropout: float = 0.1,
    ):
        super().__init__()
        self.n_corrections = n_corrections
        self.layer = nn.Linear(latent_dim + n_corrections, hidden_dim)
        self.refine = nn.Linear(hidden_dim, hidden_dim)
        self.film = FiLM(n_corrections, hidden_dim) if n_corrections else None
        self.activation = nn.LeakyReLU(negative_slope=leaky_relu_slope)
        self.dropout1 = nn.Dropout(dropout)
        self.dropout2 = nn.Dropout(dropout)
        self.apply(_orthogonal_init)

    def forward(
        self, latent: torch.Tensor, corrections: torch.Tensor | None = None
    ) -> torch.Tensor:
        """:returns: ``[samples, samples]`` edge probabilities for this batch."""
        h = latent
        if self.n_corrections:
            if corrections is None:
                raise ValueError("correction variables were expected but not supplied")
            h = torch.cat((latent, corrections.float()), dim=1)

        h = self.activation(self.layer(h))
        if self.film is not None:
            h = self.film(h, corrections)
        h = self.dropout1(h)
        h = self.dropout2(self.activation(self.refine(h)))

        return torch.sigmoid(h @ h.transpose(0, 1))

    The single-network model¶

    preffect.graph_model ¶

    The single model: counts plus a sample-sample network.

    The simple model looks at each sample alone. The single model adds a network saying which samples resemble each other, and lets a graph attention layer build each sample's representation partly from its neighbours. The hope is that a noisy profile is stabilised by the company it keeps.

    Two latent spaces, not one¶

    The model keeps a separate summary for each of the two things it is asked to reproduce: one for the counts, one for the network. Version 0.1 does the same, and the reason is that they answer different questions. A latent space good at predicting who your neighbours are need not be good at predicting your counts.

    So a forward pass runs the graph encoder twice over the same input, producing an expression summary and a network summary, and the network summary is the one the adjacency decoder reads.

    What the objective gains¶

    Two terms beyond the simple model's: how well the network is reproduced, and how far the network summary has drifted from the prior. Both only exist here, so for a simple model they are absent from the objective rather than present and zero. That distinction is why the loss is a list.

    The full model¶

    preffect.full_model ¶

    The full model: several matched tissues from the same patients.

    The single model reads one profile and its network. The full model reads several, a tumour and the normal tissue beside it for instance, matched patient by patient, and uses them together.

    What it predicts, which is narrower than it sounds¶

    It predicts the first tissue's counts only. The others inform that prediction without being predicted themselves. Version 0.1 does the same: its likelihood is built against tissue one's counts and tissue one's library size, whatever the tissue count. The paper describes this as using matched tissues to stabilise a profile, which is exactly what it is, but the phrase "multiple expression profiles" can read as though all of them were being modelled jointly. They are not.

    Every tissue does get its own network reconstructed, so the objective has one network term per tissue and one expression term in total.

    How the tissues are combined¶

    Each tissue is encoded separately, producing one expression summary and one network summary per tissue. The expression summaries are then concatenated and squeezed back down to the size of one, and that combined summary is what the count heads read. So information crosses between tissues at exactly one point, after encoding and before decoding.

    Distributions¶

    preffect.distributions ¶

    The count distributions PREFFECT fits.

    Two distributions, both for counts.

    Negative binomial. A Poisson distribution says that if a gene is expressed at some average level, the counts you see vary only as much as pure counting noise allows. Real sequencing counts vary more than that, so the negative binomial adds a second parameter for the extra spread. It arises by letting the Poisson rate itself vary from sample to sample according to a gamma distribution, which is why it is sometimes called the gamma-Poisson.

    Zero-inflated negative binomial. The same thing, plus a separate probability that a count is zero for a reason unrelated to expression level, such as the transcript never being captured. The paper that accompanies version 0.1 found little evidence that fRNA-seq needs the zero-inflated form, and the negative binomial is the default here for that reason, but both are available.

    How the parameters are named¶

    mu is the mean. theta is the inverse dispersion: large theta means counts close to Poisson, small theta means heavily overdispersed. The variance is mu + mu**2 / theta, so theta appears in the denominator, and this is the usual trap when reading the code against a textbook that calls the dispersion alpha = 1 / theta.

    zero_logits is the log-odds of an extra zero, not a probability. Working in log-odds keeps the arithmetic stable when the probability is near 0 or 1.

    Lineage¶

    The parameterisation and the arrangement of the log-probability follow scvi-tools (Gayoso, Lopez, Kim, Boyeau, Yosef and others; BSD 3-Clause), which is what version 0.1 vendored. These implementations are written from the gamma-Poisson definition rather than copied, and tests/test_distributions.py checks them against version 0.1's numbers to within floating point tolerance, so the lineage is verified rather than asserted.

    class NegativeBinomial:
    """A negative binomial over counts, parameterised by mean and inverse dispersion.

    Deliberately not a subclass of :class:`torch.distributions.Distribution`.
    The only things PREFFECT asks of it are a log probability, a mean, a
    variance and a sample, and the constraint machinery that comes with the
    base class costs more than it gives here.
    """

    def __init__(self, mu: torch.Tensor, theta: torch.Tensor, eps: float = DEFAULT_EPS):
        if mu.shape != theta.shape:
            raise ValueError(
                f"mu is shaped {tuple(mu.shape)} and theta {tuple(theta.shape)}; "
                f"they have to match"
            )
        self.mu = mu
        self.theta = theta
        self.eps = eps

    @property
    def mean(self) -> torch.Tensor:
        return self.mu

    @property
    def variance(self) -> torch.Tensor:
        """``mu + mu**2 / theta``: Poisson variance plus the extra spread."""
        return self.mu + (self.mu**2) / self.theta

    def log_prob(self, x: torch.Tensor) -> torch.Tensor:
        """Log probability of each count in ``x`` under this distribution.

        One value per element, not a sum, so the caller decides what to reduce
        over. :func:`log_prob_negative_binomial` does the arithmetic; this is
        the method form, for code that holds a distribution object.
        """
        return log_prob_negative_binomial(x, self.mu, self.theta, self.eps)

    def sample(self, sample_shape: torch.Size | tuple[int, ...] = ()) -> torch.Tensor:
        """Draw counts, through the gamma-Poisson construction it is defined by."""
        shape = torch.Size(sample_shape)
        concentration = self.theta.expand(shape + self.theta.shape)
        rate = (self.theta / self.mu).expand(shape + self.mu.shape)
        rates = torch.distributions.Gamma(concentration, rate).sample()
        return torch.poisson(rates)

    def __repr__(self) -> str:
        return f"NegativeBinomial(shape={tuple(self.mu.shape)})"

    def __init__(
        self,
        mu: torch.Tensor,
        theta: torch.Tensor,
        zero_logits: torch.Tensor,
        eps: float = DEFAULT_EPS,
    ):
        super().__init__(mu=mu, theta=theta, eps=eps)
        if zero_logits.shape != mu.shape:
            raise ValueError(
                f"zero_logits is shaped {tuple(zero_logits.shape)} and mu "
                f"{tuple(mu.shape)}; they have to match"
            )
        self.zero_logits = zero_logits

    @property
    def zero_probs(self) -> torch.Tensor:
        """The extra-zero probability, converted from log-odds."""
        return torch.sigmoid(self.zero_logits)

    @property
    def mean(self) -> torch.Tensor:
        """Lower than ``mu``, because some counts are forced to zero."""
        return (1 - self.zero_probs) * self.mu

    @property
    def variance(self) -> torch.Tensor:
        pi = self.zero_probs
        nb_variance = self.mu + (self.mu**2) / self.theta
        return (1 - pi) * (nb_variance + pi * self.mu**2)

    def log_prob(self, x: torch.Tensor) -> torch.Tensor:
        """Log probability of each count in ``x``, allowing for extra zeros.

        One value per element, not a sum. A zero can arise two ways here,
        either from the negative binomial or from the extra-zero component,
        and both routes are counted;
        :func:`log_prob_zero_inflated_negative_binomial` does the arithmetic.
        """
        return log_prob_zero_inflated_negative_binomial(
            x, self.mu, self.theta, self.zero_logits, self.eps
        )

    def sample(self, sample_shape: torch.Size | tuple[int, ...] = ()) -> torch.Tensor:
        counts = super().sample(sample_shape)
        forced_zero = torch.rand_like(counts) < self.zero_probs.expand_as(counts)
        return torch.where(forced_zero, torch.zeros_like(counts), counts)

    """Log probability of observing counts ``x``, element by element.

    :param x: observed counts.
    :param mu: mean, positive.
    :param theta: inverse dispersion, positive.
    """
    log_theta_mu = torch.log(theta + mu + eps)

        theta * (torch.log(theta + eps) - log_theta_mu)
        + x * (torch.log(mu + eps) - log_theta_mu)
        + torch.lgamma(x + theta)
        - torch.lgamma(theta)
        - torch.lgamma(x + 1)
    )

    def log_prob_zero_inflated_negative_binomial(
    x: torch.Tensor,
    mu: torch.Tensor,
    theta: torch.Tensor,
    zero_logits: torch.Tensor,
    eps: float = DEFAULT_EPS,
) -> torch.Tensor:
    """Log probability under the zero-inflated negative binomial.

    A count of zero can arise two ways, through the extra-zero process or
    through the negative binomial itself, so the zero case sums both. Any other
    count can only come from the negative binomial.

    :param zero_logits: log-odds of the extra zero process, any sign.
    """
    softplus_pi = F.softplus(-zero_logits)
    log_theta_mu = torch.log(theta + mu + eps)
    pi_theta_log = -zero_logits + theta * (torch.log(theta + eps) - log_theta_mu)

    case_zero = F.softplus(pi_theta_log) - softplus_pi
    case_non_zero = (
        -softplus_pi
        + pi_theta_log
        + x * (torch.log(mu + eps) - log_theta_mu)
        + torch.lgamma(x + theta)
        - torch.lgamma(theta)
        - torch.lgamma(x + 1)
    )

    # Selected rather than branched, so both sides stay differentiable.
    is_zero = (x < eps).to(case_zero.dtype)
    return is_zero * case_zero + (1.0 - is_zero) * case_non_zero

    Fitting both distributions to one vector of counts and comparing them, which is how the paper decided that the plain negative binomial is enough for fRNA-seq. Choosing a distribution is the page to read first.

    preffect.zinb_fit ¶

    Fitting a count distribution to one vector of counts, and saying which fits better.

    This answers a question that comes before any model is trained: do these counts need a zero-inflated distribution at all? It fits a zero-inflated negative binomial and a plain negative binomial to the same counts and reports both, with the Akaike information criterion for each so they can be compared. The criterion is the fitted log-likelihood penalised by the number of parameters, and the smaller value wins, so adding the zero-inflation parameter has to buy enough extra fit to pay for itself.

    Nothing here trains PREFFECT. This is the standalone estimator the paper that accompanies version 0.1 used to decide that the negative binomial is enough for fRNA-seq, and that conclusion is why :class:~preffect.config.Likelihood defaults to the negative binomial. It is here so the evidence can be reproduced and extended to new cohorts.

    What it reports¶

    For the zero-inflated fit: the mean of the negative binomial component, the inverse dispersion, the probability of an extra zero, the log-likelihood and the criterion. For the plain fit: the same without the zero probability. Note that the reported mean belongs to the negative binomial component, as it did in version 0.1; the mean of the zero-inflated distribution as a whole is (1 - zero_probability) * mean, which is lower.

    dispersion here is PREFFECT's theta, the inverse dispersion: large theta means counts close to Poisson, small theta means heavily overdispersed. The variance is mu + mu**2 / theta. See :mod:preffect.distributions, which this module shares its arithmetic with.

    How it differs from version 0.1¶

    Version 0.1 shipped this as wrappers/_ZINBEstimator.py. The likelihood is the same and the answers agree on counts version 0.1 could fit, but three things changed. They are recorded as findings 57, 58 and 60 in how_this_project_breaks.md with the measurements, and in the decision of 2026-11-02 on this estimator.

    1. The search runs over the mean and the dispersion, not over the mean and the variance with the dispersion divided out afterwards. Version 0.1's division could return a negative or an infinite dispersion, because nothing kept the fitted variance above the fitted mean. Here the dispersion is a parameter with a positivity bound, so there is nothing to divide.
    2. Counts whose variance falls below their mean are fitted, rather than silently returning nothing. Version 0.1 started every one of its eight searches at the sample mean and variance, which for such counts is outside the model, so no search moved and the caller got None.
    3. A fit that cannot be made raises, naming the counts and the reason, instead of returning None for a caller to notice or not.

    The gradient comes from :mod:preffect.distributions by automatic differentiation rather than being estimated by finite differences, which is why the search needs about a fifth as many evaluations of the likelihood.

    def fit_negative_binomial(counts) -> NegativeBinomialFit:
    """Fit a negative binomial to one vector of counts by maximum likelihood.

    This is the companion the zero-inflated fit is compared against. The
    comparison only means something if both sides are fitted the same way, so
    this maximises the same likelihood with the same search rather than reading
    the mean and variance off the data, which is what the simplest of the
    paper's notebooks did. The difference that makes is measured in the
    decision of 2026-11-02 on this estimator: under a tenth of a criterion unit
    on counts without extra zeros, and 230 units on counts with them.

    :param counts: one vector of whole, non-negative numbers, one per sample.
    :raises DataError: the counts are not counts.
    :raises FitError: every count is zero, or no search converged.
    """
    values = _as_counts(counts)
    _refuse_if_every_count_is_zero(values)
    tensor = torch.tensor(values, dtype=torch.float64)

    start_mean, start_dispersion = _starting_point(
        float(values.mean()), float(values.var()), 0.0
    )
    result = minimize(
        _negative_log_likelihood_and_gradient,
        np.array([math.log(start_mean), math.log(start_dispersion)]),
        args=(tensor, False),
        jac=True,
        method="L-BFGS-B",
        bounds=_log_bounds(zero_inflated=False),
    )
    if not _has_arrived(result):
        raise FitError(
            f"the negative binomial fit did not converge on {values.size} counts "
            f"with mean {values.mean():g} and variance {values.var():g}; the "
            f"search reported: {result.message}"
        )

    mean = float(np.exp(result.x[0]))
    dispersion = float(np.exp(result.x[1]))
    log_likelihood = -float(result.fun)
    return NegativeBinomialFit(
        mean=mean,
        dispersion=dispersion,
        log_likelihood=log_likelihood,
        aic=2 * 2 - 2 * log_likelihood,
        dispersion_at_bound=_at_dispersion_bound(dispersion),
    )

    def fit_zero_inflated_negative_binomial(
    counts,
    starting_zero_probabilities: tuple[float, ...] = DEFAULT_STARTING_ZERO_PROBABILITIES,
) -> ZeroInflatedFit:
    """Fit a zero-inflated negative binomial to one vector of counts.

    One search is run from each starting zero-inflation probability and the best
    fit is kept, because the likelihood of heavily zero-inflated counts has a
    second peak where the extra zeros are explained as overdispersion instead.

    :param counts: one vector of whole, non-negative numbers, one per sample.
    :param starting_zero_probabilities: the probabilities to start from, one
        search each. Defaults to version 0.1's eight values.
    :raises DataError: the counts are not counts, or a starting probability is
        not strictly between zero and one.
    :raises FitError: every count is zero, or no search converged.
    """
    values = _as_counts(counts)
    _refuse_if_every_count_is_zero(values)
    if not starting_zero_probabilities:
        raise DataError("no starting zero-inflation probabilities were given")
    for probability in starting_zero_probabilities:
        if not 0.0 < probability < 1.0:
            raise DataError(
                f"a starting zero-inflation probability has to lie strictly "
                f"between 0 and 1; got {probability!r}"
            )

    tensor = torch.tensor(values, dtype=torch.float64)
    sample_mean = float(values.mean())
    sample_variance = float(values.var())
    bounds = _log_bounds(zero_inflated=True)

    best = None
    converged = 0
    last_message = "no search was run"
    for probability in starting_zero_probabilities:
        start_mean, start_dispersion = _starting_point(
            sample_mean, sample_variance, probability
        )
        start = np.array(
            [
                math.log(start_mean),
                math.log(start_dispersion),
                math.log(probability / (1.0 - probability)),
            ]
        )
        result = minimize(
            _negative_log_likelihood_and_gradient,
            start,
            args=(tensor, True),
            jac=True,
            method="L-BFGS-B",
            bounds=bounds,
        )
        last_message = str(result.message)
        if not _has_arrived(result):
            continue
        converged += 1
        if best is None or result.fun < best.fun:
            best = result

    if best is None:
        raise FitError(
            f"none of the {len(starting_zero_probabilities)} searches converged "
            f"on {values.size} counts with mean {sample_mean:g} and variance "
            f"{sample_variance:g}; the last one reported: {last_message}"
        )

    mean = float(np.exp(best.x[0]))
    dispersion = float(np.exp(best.x[1]))
    zero_probability = float(1.0 / (1.0 + np.exp(-best.x[2])))
    log_likelihood = -float(best.fun)
    return ZeroInflatedFit(
        mean=mean,
        dispersion=dispersion,
        zero_probability=zero_probability,
        log_likelihood=log_likelihood,
        aic=2 * 3 - 2 * log_likelihood,
        dispersion_at_bound=_at_dispersion_bound(dispersion),
        starts_converged=converged,
    )

            f"expected a matrix of samples by genes, got an array of shape "
            f"{tuple(values.shape)}; for one vector of counts use compare_fits"

        names = [str(index) for index in range(values.shape[1])]

            f"got {len(names)} names for {values.shape[1]} columns; they have to match"
        )

    rows = []
    for index, name in enumerate(names):
        column = values[:, index]
